DROP DATABASE IF EXISTS sistema_gerencial;
CREATE DATABASE sistema_gerencial
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE sistema_gerencial;

CREATE TABLE endereco (
    id INT NOT NULL AUTO_INCREMENT,
    cep VARCHAR(12) NOT NULL,
    rua VARCHAR(255) NOT NULL,
    numero INT NOT NULL,
    complemento VARCHAR(255) NULL,
    bairro VARCHAR(255) NOT NULL,
    cidade VARCHAR(255) NOT NULL,
    uf CHAR(2) NOT NULL,
    PRIMARY KEY (id)
);

CREATE TABLE dados_login (
    id INT NOT NULL AUTO_INCREMENT,
    login VARCHAR(255) NOT NULL,
    senha VARCHAR(255) NOT NULL,
    perfil VARCHAR(4) NOT NULL,
    PRIMARY KEY (id),
    UNIQUE KEY uq_dados_login_login (login)
);

CREATE TABLE contato (
    id INT NOT NULL AUTO_INCREMENT,
    telefone VARCHAR(15) NOT NULL,
    celular VARCHAR(15) NOT NULL,
    email VARCHAR(100) NOT NULL,
    PRIMARY KEY (id),
    UNIQUE KEY uq_contato_email (email)
);

CREATE TABLE usuario (
    id INT NOT NULL AUTO_INCREMENT,
    nome VARCHAR(255) NOT NULL,
    cpf VARCHAR(11) NOT NULL,
    data_nascimento DATE NOT NULL,
    sexo CHAR(1) NOT NULL,
    dados_login_id INT NOT NULL,
    endereco_id INT NOT NULL,
    contato_id INT NOT NULL,
    PRIMARY KEY (id),
    UNIQUE KEY uq_usuario_cpf (cpf),
    CONSTRAINT fk_usuario_endereco FOREIGN KEY (endereco_id) REFERENCES endereco(id),
    CONSTRAINT fk_usuario_dados_login FOREIGN KEY (dados_login_id) REFERENCES dados_login(id),
    CONSTRAINT fk_usuario_contato FOREIGN KEY (contato_id) REFERENCES contato(id)
);

CREATE TABLE produtos (
    id INT NOT NULL AUTO_INCREMENT,
    nome VARCHAR(255) NOT NULL,
    preco_custo DECIMAL(10,2) NOT NULL,
    preco_venda DECIMAL(10,2) NOT NULL,
    quantidade INT NOT NULL,
    medida VARCHAR(50) NOT NULL,
    categoria VARCHAR(50) NOT NULL,
    PRIMARY KEY (id)
);

CREATE TABLE vendas (
    id INT NOT NULL AUTO_INCREMENT,
    id_usuario INT NOT NULL,
    data_venda DATETIME NOT NULL,
    total DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT fk_vendas_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id)
);

CREATE TABLE lista_venda (
    id INT NOT NULL AUTO_INCREMENT,
    id_venda INT NOT NULL,
    id_produto INT NOT NULL,
    quantidade INT NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT fk_lista_venda_venda FOREIGN KEY (id_venda) REFERENCES vendas(id),
    CONSTRAINT fk_lista_venda_produto FOREIGN KEY (id_produto) REFERENCES produtos(id)
);
